(начало кода остается прежним до класса Bin) class Bin { constructor(cfg, i, total) { this.cfg = cfg; this.scale = 1; this.updatePos(i, total); } updatePos(i, total) { // АДАПТИВНОСТЬ: Баки не должны быть выше 25% высоты экрана let maxH = H * 0.25; let gap = W / total; // Ширина бака зависит от количества, но ограничена высотой this.w = Math.min(gap * 0.85, maxH * 0.8, 120); this.h = this.w * 1.25; this.x = i * gap + (gap - this.w) / 2; // Прижимаем к самому низу this.y = H - this.h - 5; } draw() { let img = Loader.imgs['b_'+this.cfg.id]; if(!img) return; let sw = this.w * this.scale, sh = this.h * this.scale; ctx.drawImage(img, this.x - (sw-this.w)/2, this.y - (sh-this.h)/2, sw, sh); if(this.scale > 1) this.scale -= 0.05; } } // ... (внутри функции spawnAnimal) function spawnAnimal() { // Появление в верхней части экрана (сбоку от солнца) const side = Math.random() > 0.5 ? 1 : -1; const spawnX = W/2 + (side * W * 0.3); gameState.animal = { img: Loader.imgs['a'+(Math.floor(Math.random()*10)+1)], x: spawnX, y: H * 0.2, // Поднимаем выше, чтобы не закрывать дорогу life: 2.5, s: 0 }; } // ... (остальной код функции resize должен вызывать updatePos для всех баков) function resize() { W = window.innerWidth; H = window.innerHeight; canvas.width = W; canvas.height = H; let ks = Object.keys(TYPES); if (gameState.bins) { gameState.bins.forEach((b, i) => b.updatePos(i, ks.length)); } }
